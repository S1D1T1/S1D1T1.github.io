func createAlbum(named title: String, with assets: [PHAsset]) {
    PHPhotoLibrary.shared().performChanges {

Claude:
The iPhone lens identification requires parsing EXIF data and mapping it to known iPhone camera configurations. Here's how we'd determine the lens type:
swift
enum LensType {
    case ultraWide     // 0.5x
    case wide         // 1x
    case telephoto2x  // 2x
    case telephoto3x  // 3x
    case unknown
    
    var multiplier: String {
        switch self {
        case .ultraWide: return "0.5x"
        case .wide: return "1x"
        case .telephoto2x: return "2x"
        case .telephoto3x: return "3x"
        case .unknown: return "unknown"
        }
    }
}

func determineLensType(from asset: PHAsset) async throws -> LensType {
    // Get resource
    let resources = PHAssetResource.assetResources(for: asset)
    guard let resource = resources.first else { throw LensError.noResource }
    
    // Several pieces of metadata we can check:
    
    // 1. FocalLength - most reliable
    // iPhone models vary slightly but generally:
    // - Ultra Wide: 13-14mm
    // - Wide: 23-26mm
    // - Telephoto 2x: 48-52mm
    // - Telephoto 3x: 77mm
    if let focalLength = getExifFocalLength(from: resource) {
        switch focalLength {
        case 13...14: return .ultraWide
        case 23...26: return .wide
        case 48...52: return .telephoto2x
        case 77...78: return .telephoto3x
        default: return .unknown
        }
    }
    
    // 2. Lens model string - backup method
    // Format varies but often contains useful info:
    // "iPhone 13 Pro back ultra wide camera 13mm f/2.4"
    // "iPhone 14 Pro back camera 3x telephoto 77mm f/2.8"
    if let lensModel = getExifLensModel(from: resource) {
        if lensModel.contains("ultra wide") { return .ultraWide }
        if lensModel.contains("telephoto 3x") { return .telephoto3x }
        if lensModel.contains("telephoto") { return .telephoto2x }
        if lensModel.contains("wide") { return .wide }
    }
    
    return .unknown
}
The reference values come from:
iPhone camera specifications (published by Apple)
Sample photos from different iPhone models
EXIF data patterns from different iPhone generations

Future iPhone models might introduce new lens types

func getExifFocalLength(from asset: PHAsset) async throws -> Double? {
    // Request image data and metadata
    let resources = PHAssetResource.assetResources(for: asset)
    guard let resource = resources.first else { return nil }
    
    return await withCheckedContinuation { continuation in
        let options = PHContentEditingInputRequestOptions()
        asset.requestContentEditingInput(with: options) { input, info in
            guard let url = input?.fullSizeImageURL,
                  let source = CGImageSourceCreateWithURL(url as CFURL, nil),
                  let metadata = CGImageSourceCopyPropertiesAtIndex(source, 0, nil) as? [String: Any],
                  let exif = metadata[kCGImagePropertyExifDictionary as String] as? [String: Any],
                  let focalLength = exif[kCGImagePropertyExifFocalLength as String] as? Double 
            else {
                continuation.resume(returning: nil)
                return
            }
            
            continuation.resume(returning: focalLength)
        }
    }
}

Request the ContentEditingInput to get access to the actual image data
Use CoreGraphics' image source APIs to read the metadata
Navigate through the EXIF dictionary to find the focal length
